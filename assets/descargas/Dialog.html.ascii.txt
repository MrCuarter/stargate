// v3.30 - El calendario ya no se pregunta: sale entero de la fecha de la semana 1. Los formularios
// abren ese mismo dia, el registro de misiones cierra al acabar la ultima semana y el canje aguanta
// una semana mas. Aqui solo se cuenta, para que el docente lo vea antes de crear nada.
(function(){
  var SEM = {"REGULAR":15,"PUA":8}, EXTRA = 1;
  function iso(d){ var p=function(n){return (n<10?'0':'')+n;}; return d.getFullYear()+'-'+p(d.getMonth()+1)+'-'+p(d.getDate()); }
  function mas(base, dias){ var d = new Date(base.getTime()); d.setDate(d.getDate()+dias); return d; }
  function bonito(s){ if(!s) return '\u2014'; var p = s.split('-'); return p[2]+'/'+p[1]+'/'+p[0]; }
  var elIni = document.getElementById('inicio');
  // el lunes que viene a las 00:00 (si hoy es lunes, hoy): es cuando empiezan las semanas del curso
  var hoy = new Date(); hoy.setHours(0,0,0,0);
  if (elIni && !elIni.value) elIni.value = iso(mas(hoy, (8 - (hoy.getDay() || 7)) % 7));
  function recalcular(){
    var nota = document.getElementById('calNota'); if (!nota) return;
    var ini = elIni ? elIni.value : '';
    var tipo = document.getElementById('tipo').value === 'PUA' ? 'PUA' : 'REGULAR';
    var n = SEM[tipo];
    if (!ini) { nota.textContent = 'Pon la fecha en la que empieza la semana 1 y el calendario se calcula solo.'; return; }
    var base = new Date(ini + 'T00:00:00'); if (isNaN(base.getTime())) return;
    nota.innerHTML = '<b>' + tipo + ' \u00b7 ' + n + ' semanas.</b> Los tres formularios abren el <b>' + bonito(ini) +
      '</b>, el registro de misiones cierra el <b>' + bonito(iso(mas(base, n*7-1))) +
      '</b> y el canje de recompensas el <b>' + bonito(iso(mas(base, (n+EXTRA)*7-1))) +
      '</b>, a las 23:59. Se puede mover despu\u00e9s desde el panel de profes.';
  }
  ['inicio','tipo'].forEach(function(k){ var el = document.getElementById(k);
    if (el) { el.addEventListener('change', recalcular); el.addEventListener('input', recalcular); } });
  recalcular();
})();
