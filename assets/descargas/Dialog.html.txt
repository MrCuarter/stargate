function v(id){return document.getElementById(id).value;}
var nFilas=0;
function filaDocente(){
  if(nFilas>=10){document.getElementById('mas').disabled=true;return;}
  nFilas++;
  var tr=document.createElement('tr'); tr.setAttribute('data-fila',nFilas);
  tr.innerHTML='<td style="padding:2px 4px 2px 0"><input class="dn" placeholder="Nombre Apellido"></td>'
    +'<td style="padding:2px 0"><input class="dc" type="email" placeholder="correo@unir.net"></td>'
    +'<td style="text-align:center"><input class="dr" type="radio" name="ref" style="width:auto"'+(nFilas===1?' checked':'')+'></td>';
  document.getElementById('docentes').appendChild(tr);
  if(nFilas>=10) document.getElementById('mas').disabled=true;
}
document.addEventListener('DOMContentLoaded',function(){
  for(var i=0;i<3;i++) filaDocente();
  document.getElementById('mas').onclick=filaDocente;
});
function crear(){
  var docentes=[],ref='';
  Array.prototype.forEach.call(document.querySelectorAll('#docentes tr[data-fila]'),function(tr){
    var n=tr.querySelector('.dn').value.trim(), c=tr.querySelector('.dc').value.trim();
    if(!n) return;
    var esRef=tr.querySelector('.dr').checked;
    if(esRef&&!ref) ref=n;
    docentes.push({nombre:n,correo:c,rol:esRef?'referente':'imparte'});
  });
  var malos=docentes.filter(function(x){return x.correo&&!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(x.correo);});
  if(malos.length){document.getElementById('err').textContent='Revisa el correo de '+malos[0].nombre;return;}
  var d={nombre:v('nombre'),tipo:v('tipo'),inicio:v('inicio'),referente:ref,
         profesores:docentes.filter(function(x){return x.rol!=='referente';}).map(function(x){return x.nombre;}).join(', '),
         docentes:docentes,apertura:v('apertura'),cierre:v('cierre'),panelVer:v('panelVer'),panelEdit:v('panelEdit')};
  if(!d.nombre){document.getElementById('err').textContent='Pon el nombre del PER';return;}
  if(!docentes.length){document.getElementById('err').textContent='Añade al menos un docente';return;}
  document.getElementById('go').disabled=true;document.getElementById('err').textContent='Creando los 3 formularios y las pestañas (30-60 s)…';
  google.script.run.withSuccessHandler(ok).withFailureHandler(function(e){document.getElementById('err').textContent=e.message;document.getElementById('go').disabled=false;}).crearPER(d);
}

  function b(t,u){return '<div class="ok"><b>'+t+'</b><br><a href="'+u+'" target="_blank">'+u+'</a></div>';}
  x.innerHTML='<h3>PER «'+r.nombre+'» ('+r.tipo+') creado · '+r.estado+'</h3>'
  +(r.restos?'<div class="ok"><b>Aviso:</b> había '+r.restos+' pestaña(s) de un PER anterior con el mismo nombre. No se han borrado: se han renombrado a «restos · …». Si no las necesitas, límpialas con el menú STARGATE → Mantenimiento → Limpiar restos de PER borrados.</div>':'')
  +b('Bitácora de mando (alumnado)',r.formBitacora)+b('Ticket de salida (anónimo)',r.formTicket)+b('Canje de recompensas',r.formCanje)
  +(r.doc?'<div class="destacado"><b>Todo el PER en un documento</b><small>Enlaces, códigos para incrustar y QR, listo para compartir con tu profesorado</small><a href="'+r.doc+'" target="_blank">ABRIR EL DOCUMENTO DE ENLACES</a></div>':'')
  +(r.nave?b('La Nave del Recluta (la web del alumnado: onboarding, semanas, estado y recompensas)',r.nave):'')
  +(r.panelVer?b('Panel de control (visualización, para el alumnado)',r.panelVer):'<div class="ok"><b>Panel de control:</b> sin definir. Ponlo en la hoja (menú STARGATE → Guardar panel de control estándar) o en Ajustes del PER.</div>')
  +(r.panelEdit?b('✏️ Panel de control (edición, para el profesorado)',r.panelEdit):'')
+b('Generador de embeds en la web (elige tu nombre)','https://stargate.mistercuarter.es/embed.html?per='+r.id)+b('Tablero del PER en la web',r.web)+b('Foro dinámico del PER',r.foro)
  +(r.embedNave?'<div class="ok"><b>Embed de la Nave del Recluta para el Genially del PER</b><br><textarea class="emb" readonly>'+r.embedNave.replace(/</g,'&lt;')+'</textarea></div>':'')

}
